%  ※ \SecA{} 等の {} 内に % を書くとコメントになりエラーが出ます
%  見出し番号: \SecZ=番号なし \SecA=第１ \SecB=１ \SecC=⑴ \SecD=ア \SecE=(ア)
%  ※ 見出し（\SecA{} 等）は本文なしで連続させると番号表示が崩れるため、

% 事件番号は「R８(わ)３２」の形式で記載する（「令和」「年」「第」「号」は書かない）
\Idcs{
  [o]{R８(わ)　　:【罪名】被告事件\\被告人:【氏/名】:}
}

% 標題は書面名に置き換える（例：弁論要旨、冒頭陳述書、保釈請求書の添付書面 など）
\Tit{【弁論要旨】}
\Dat{令和８年　　月　　日}
\Recs{
  [n]{:【裁判所名】【支部名】支部:御中}
}

% 差出人欄の横幅：\senderhlength{①}{②}{③}{④}
%   ①左端からのインデント（字下げ）  ②肩書の幅  ③肩書と名前の間隔  ④名前の幅
% 既定値は {13}{10}{2}{10}。肩書の幅が既定の10字だと、「弁護人」（3字）のような
% 短い肩書は均等割付けで「弁　　護　　人」と間延びするため、肩書の幅を字数＋１字程度に詰め、
% 詰めた分だけ字下げを増やして名前の位置を既定に近づけている。
% 肩書を変えるときの目安：
%   弁護人（3字）        → {17}{4}{2}{10}（この値）
%   主任弁護人（5字）    → {15}{6}{2}{10}
%   弁護人　弁護士（6字） → {14}{7}{2}{10}
\senderhlength{17}{4}{2}{10}
\Sens{
  [0.5]{%
    弁護人:【氏名】\\
  }
}
